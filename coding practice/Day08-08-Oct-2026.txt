  hosts: all
  become: yes

  tasks:

  	  	state: present


  	- name: Install Docker Compose

  	    name: docker-compose
  	    state: present
  	    executable: pip3

 # TERRAFORM - EC2 with Security Group

 resource "aws_security_group" "allow_ssh_http" {
 	name 			= "allow_ssh_http"
 	description 	= "Allow SSH and HTTP access"
 	vpc_id 			= var.vpc_id

 	ingress {
 		description = "SSH from my IP"
 		from_port 	= 22
 		to_port 	= 22
 		protocol 	= "tcp"
 		cidr_block  = [var.my_ip_cidr]
 	}

 	ingress {
 		description = "HTTP access"
 		from_port	= 80
 		to_port		= 80
 		protocol 	= "tcp"
 		cidr_blocks = ["0.0.0.0/0"]
 	}

 	egress {
 		from_port 	= 0
 		to_port		= 0
 		protocol 	= "-1"
 		cidr_blocks =["0.0.0.0/0"]
 	}
 }

# JENKINS - Docker Build and Push pipeline

pipeline {
	agent any

	environment {
		DOCKERHUB_REPO = 'your-dockerhub-username/pathnex-app'
	}

	stages{
		stage('Build Docker Image'){
			steps{
				script {
					docker.build("${DOCKERHUB_REPO}:${BUILD_NUMBER}")
				}
			}
		}
		stage('Push Docke Image'){
			steps {
				script{
					docker.withRegistry(
						'https://index.docker.io/v1/',
						'docker-credentials'
					) {
						docker.image(
							"${DOCKERHUB_REPO}:${BUILD_NUMBER}"
						).push()

						docker.image(
							"${DOCKERHUB_REPO}:${BUILD_NUMBER}"
						).push('latest')
					}
				}
			}
		}
	}
}

# GITLAB - Build and Push


stages:
	- build
	- push

variables:
	IMAGE_TAG: "$CI_REGISTRY_IMAGE:$CI_COMMIT_SHA"

build:
	stage: build
	script:
		- docker build -t "$IMAGE_TAG" .

push:
	stage: push
	script:
		- echo "$CI_REGISTRY_PASSWORD" | docker login "$CI_REGISTRY" -u "$CI_REGISTRY_USER" --password-stdin
		- docker push "$IMAGE_TAG"
