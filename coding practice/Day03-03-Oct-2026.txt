#################################################
# ANSIBLE - Install Apache and start service 

--- 

- name: Install and Start Apache service 
  hosts: all
  become: yes

  tasks:
    - name: Install httpd
      ansible.builtin.yum:
        name: httpd 
        state: present 
      
    - name: Start Apache service 
      ansible.builtin.service:
        name: httpd 
        state: started 
        enabled: yes
