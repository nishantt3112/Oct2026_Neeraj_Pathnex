# ANSIBLE - Setup Nginx on multiple hosts

- name: Setup Nginx on multiple nodes 
  hosts: all
  become: yes

  tasks:
  	- name: Install Nginx 
  	  ansible.builtin.yum:
  	  	name: nginx 
  	  	state: present

  	- name: Start and enable Nginx
  	  ansible.builtin.service:
  	  	name: nginx
  	  	state: started
  	  	enabled: yes

# TERRAFORM - Create IAM Users and Policies

terraform {

	required_version = "= 1.9.0"  
	required_providers {
	aws = {
		source = "hashicorp/aws"
		version = "~> 6.0"
	} 
	}
}

provider "aws" {
	region = "us-east-1"
}


resource "aws_iam_user" "example_user" {
	name = "pathnex_user"
}

resource "aws_iam_policy" "examle_policy" {
	name = "pathnex-policy"
	description = "Policy for Pathnex application"

	policy = jsonencode({
		Version = "2012-10-17"
		Statement = [
		{
			Action = [
			"s3:ListBucket",]
			Effect = "Allow"
			Resource = "arn:aws:s3:::pathnex-bucket"
		}
		]
	})
}

# KUBERNETES -  Deploy Nginx with ConfigMap and Secrets 

---
apiVersion: v1
kind: ConfigMap
metadata:
	name: nginx-config
data:
	nginx.conf: |
		server {
			listen 80;
			server_name localhost;

			location / {
				root /user/share/nginx/html;
				index index.html index.htm;
			}
		}

---

apiVersion: v1
kind: Secret
metadata:
	name: nginx-secret
type: Opaque
stringData:
	password: "replace-with-a-real-secret"

---

apiVersion: apps/v1
kind: Deployment
metadata:
	name: pathnex-nginx
spec:
	replicas: 2
	selector:
		matchLabels:
			app: pathnex-nginx
	template:
		metadata:
			labels:
				app: pathnex-nginx
		spec:
			containers:
				- name: nginx
				  image: nginx:stable
				  ports:
				  	- containerPort: 80
				  volumeMounts:
				  	- name: nginx-config
				  	  mountPath: /etc/nginx/conf.d/default.conf
				  	  readOnly: true
				  	- name: nginx-secret
				  	  mountPath: /run/secrets/nginx
				  	  readOnly: true

			volumes:
				- name: nginx-config
				  configMap:
				  	name: nginx-config
				- name: nginx-secret
				  secret: 
				  	secretName: nginx-secret

# JENKINSFILE - multibranch pipeline with docker build 

pipeline {
	agent any
	environment {

	}
	stages{
		stage('Build'){
			steps {
				script {
					docker.build('pathnex-app')
				}
			}
		}
		stage('Deploy'){
			steps {
				script {
					sh 'kubectl apply -f kubernetes/deployment.yaml'
				}
			}
		}

	}
}

# GITLAB - Advanced Deployment with Helm


stages:
	- build
	- push
	- deploy

variables:   
	IMAGE_TAG: "$CI_REGISTRY_IMAGE:$CI_COMMIT_SHA"

build: 
	stage: build
	script:
		- docker build -t "$IMAGE_TAG"

push:
	stage: push
	script: 
		- echo "$CI_REGISTRY_PASSWORD" | docker login "$CI_REGISTRY" -u "$CI_REGISTRY_USER" --password-stdin
   	    - docker push "$IMAGE_TAG"

deploy: 
	stage: deploy
	script:
		- helm repo add pathnex https://charts.pathnex.com
		- helm repo update
		- helm upgrade --install pathnex-nginx pathnex/nginx-ingress
